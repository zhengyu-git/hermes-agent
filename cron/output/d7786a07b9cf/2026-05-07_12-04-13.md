```
# 执行以下 Python 代码获取新闻标题
import requests, re

headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'}

def fetch_titles(url):
    try:
        r = requests.get(url, headers=headers, timeout=15)
        r.encoding = 'utf-8'
        text = r.text
        # 移除script/style
        text = re.sub(r'<script[^>]*>.*?</script>', '', text, flags=re.DOTALL)
        text = re.sub(r'<style[^>]*>.*?</style>', '', text, flags=re.DOTALL)
        # 提取标题
        titles = re.findall(r'<a[^>]+href=[^>]+>([^<]{10,80})</a>', text)
        # 过滤导航词和短标题
        nav = {'首页','新闻','财经','体育','娱乐','军事','科技','社会','视频','图片','更多','相关','推荐'}
        titles = [t.strip() for t in titles if t.strip() not in nav and 10 < len(t.strip()) < 80]
        return titles[:30]
    except:
        return []

# 人民网
people = fetch_titles('http://www.people.com.cn/')
# 新华网
xinhua = fetch_titles('http://www.news.cn/')
# 中国新闻网
chinanews = fetch_titles('https://www.chinanews.com/')
# 澎湃新闻
thepaper = fetch_titles('https://www.thepaper.cn/')

cctv = fetch_titles('https://news.cctv.com/')
```
